> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 미분류로 쌓인 명함 약 5,000장을, 회사명 키워드 규칙으로 미리보기한 뒤 한 번에 폴더로 옮기고, 잘못되면 통째로 되돌린다.

**Architecture:** `pu-cards.html` 한 파일에 세 겹으로 넣는다 — ①순수 로직(규칙 매칭·대상 고르기, 전역 상태를 안 쓰므로 Node에서 그대로 검사 가능) ②저장 계층(`Store` 에 규칙 읽고 쓰기 추가, Firebase RTDB `pucards/classifyRules`·`pucards/classifyLog`) ③화면(환경설정 → 🧹정리 탭 안에 인라인, 팝업 없음). 적용은 명함을 하나씩 쓰지 않고 **한 번의 `update()` 다중 경로 쓰기**로 몰아 보낸다.

**Tech Stack:** 빌드 없는 단일 HTML + 바닐라 JS · Firebase RTDB(compat) · 검사는 `node --test` + `vm` 샌드박스

- CI(`node --test tests/*.test.js`)가 `deploy-pages.yml` 의 배포를 막는다. **새 검사가 통과해야 사이트가 나간다.**

```js
/* 규칙이 하나도 없을 때만 기본 4개를 심는다.
   기록 이름(key)을 고정으로 써서, 두 사람이 동시에 심어도 같은 4개로 모인다. */
function seedClassifyRules(){
  try{
    if(Store.mode!=='firebase') return;
    if(Object.keys(state.rules||{}).length) return;
    CLASSIFY_DEFAULTS.forEach(r=>{ Store.putRule({...r, words:r.words.slice()}); });
  }catch(e){ console.warn('[seedClassifyRules]', e&&e.message); }
}
/* 지금 쓸 규칙 — 저장된 게 있으면 그것, 없으면 기본값. 항상 order 오름차순 */
function classifyRuleList(){
  const saved = Object.values(state.rules||{}).filter(r=>r&&r.key);
  const list = saved.length ? saved : CLASSIFY_DEFAULTS.map(r=>({...r, words:r.words.slice()}));
  return list.slice().sort((a,b)=>(a.order||0)-(b.order||0));
}
```

```js
  const i = src.indexOf(a);
  if(i < 0) throw new Error('시작 표식 못찾음: ' + a);
  const j = src.indexOf(b, i);
  if(j < 0) throw new Error('끝 표식 못찾음: ' + b);

}
```

```js
/* 이 명함이 이 규칙에 걸리는가 — 회사명과 이름에서 단어를 찾는다.
   회사 칸이 비고 이름 칸에 상호가 들어온 명함이 많아 둘 다 본다. */
function ruleHit(it, r){
  if(!it || !r || r.enabled===false) return false;
  const hay = String(it.company||'') + ' ' + String(it.name||'');
  const ws = Array.isArray(r.words) ? r.words : [];
  for(let i=0;i<ws.length;i++){
    const w = String(ws[i]||'').trim();
    if(w && hay.indexOf(w) >= 0) return true;
  }
  return false;
}
/* 규칙별로 옮길 명함을 고른다.
   ① 미분류만 — 손으로 넣어 둔 폴더는 절대 안 건드린다
   ② 지금 보는 종류(명함/사업자)만
   ③ 위에서 아래로 먼저 맞은 규칙 하나만 (한 장이 두 폴더로 갈 수 없다) */
function classifyTargets(items, rules, kind){
  const on = (rules||[]).filter(r=>r && r.key && r.enabled!==false)
                        .slice().sort((a,b)=>(a.order||0)-(b.order||0));
  const byKey = {}; on.forEach(r=>{ byKey[r.key] = []; });
  const unmatched = [];
  (items||[]).forEach(it=>{
    if(!it || !it.id) return;
    if(((it.kind==='biz') ? 'biz' : 'card') !== kind) return;
    if(String(it.group||'') !== '') return;
    let hit = null;
    for(let i=0;i<on.length;i++){ if(ruleHit(it, on[i])){ hit = on[i]; break; } }
    if(hit) byKey[hit.key].push(it); else unmatched.push(it);
  });
  return { byKey, unmatched, keys: on.map(r=>r.key) };
}
/* 규칙이 보낼 폴더 찾기 — id 가 아니라 이름으로 가리킨다.
   폴더를 지웠다 다시 만들어도 규칙이 살아남게 하려는 것이다 (autoFolder 의 ensure 와 같은 뜻).
   잠긴 폴더로는 보내지 않는다 — 남의 개인 폴더에 명함이 섞이면 되돌리기도 어렵다. */
function findRuleGroup(groups, r, kind){
  const nm = _canon(r && r.toGroupName);
  const g = Object.values(groups||{}).find(x =>
    x && _canon(x.name) === nm && ((x.kind||'card') === kind));
  if(!g) return { gid:null, missing:true };
  if(g.locked) return { gid:null, locked:true };
  return { gid:g.id };
}
```

```js
    <button class="btn ghost" style="flex:none;padding:8px 14px"
      onclick="undoClassify('${last.id}')">되돌리기</button></div>`;
}
/* 되돌리기 — 기록해 둔 '원래 폴더'로 그대로 되돌린다. 이것도 통신 한 번. */
async function undoClassify(batchId){
  if(Store.mode!=='firebase') return toast('클라우드에 연결된 상태에서만 됩니다');
  const log = (_classifyLog||{})[batchId];
  if(!log || !log.from) return toast('되돌릴 기록을 찾지 못했습니다');
  const ids = Object.keys(log.from);
  if(!confirm(`${ids.length.toLocaleString()}장을 분류하기 전 상태로 되돌립니다.\n\n진행할까요?`)) return;
  const now = Date.now();
  const upd = {};
  ids.forEach(id=>{
    const it = state.items[id]; if(!it) return;      /* 그 사이 지워졌으면 건너뛴다 */
    const back = String(log.from[id]||'');
    upd[`items/${id}/group`]     = back;
    upd[`items/${id}/updatedAt`] = now;
    upd[`idx/${id}`] = idxRecord({ ...it, group:back });
  });
  upd[`classifyLog/${batchId}`] = null;
  toast('되돌리는 중…', 30000);
  try{
    await Store.db.ref(DB_ROOT).update(upd);
    toast(`↩️ ${ids.length.toLocaleString()}장을 되돌렸습니다`, 6000);
    setTimeout(()=>_refresh(openClassifyRules), 400);
  }catch(e){
    toast('❌ 되돌리기 실패: ' + ((e&&e.message)||'알 수 없음'), 6000);
  }
}
```
