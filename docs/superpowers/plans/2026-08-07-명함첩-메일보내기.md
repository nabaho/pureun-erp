# 명함첩 → 다음메일 보내기 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 선택한 명함의 이메일을 다음메일에 붙여넣을 수 있게 복사하고, 다음 주소록에 가져올 CSV 로 내보낸다. 수신거부한 사람은 두 길 모두에서 자동으로 빠진다.

**Architecture:** 거르는 규칙을 순수 함수 하나(`mailTargets`)에 모아 복사·내보내기·단체메일이 모두 그것만 쓴다. 두 곳에 따로 두면 한쪽만 고쳐져 수신거부한 사람에게 메일이 나간다. 수신거부는 명함의 `noMail` 과 공유 목록 `pucards/config/mailBlock` 두 갈래를 한 함수 안에서 합쳐 판단한다.

**Tech Stack:** 빌드 없는 단일 HTML + 바닐라 JS · Firebase RTDB(compat) · 검사는 `node --test` + `vm` 샌드박스

/* ══════ 메일 보내기 — 순수 로직 (테스트 대상) ══════
   거르는 규칙을 여기 한 곳에만 둔다. 복사·내보내기·단체메일이 모두 이것만 쓴다.
   두 곳에 나눠 두면 한쪽만 고쳐져 수신거부한 사람에게 메일이 나간다. */
const _MAIL_RE = /[A-Za-z0-9._%+\-]+@[A-Za-z0-9.\-]+\.[A-Za-z]{2,}/g;
/* 비교는 언제나 소문자·공백제거로. 대소문자만 달라도 거부가 안 걸리면 또 보내게 된다. */
function normEmail(v){
  const s = String(v==null?'':v).trim().toLowerCase();
  return /^[a-z0-9._%+\-]+@[a-z0-9.\-]+\.[a-z]{2,}$/.test(s) ? s : '';
}
/* 다음메일에서 긁어온 글에는 이름·날짜가 섞여 있다. 이메일처럼 생긴 것만 뽑는다.
   형식을 맞춰 달라고 하면 손으로 다듬게 되고, 그러면 안 쓰게 된다. */
function pickEmails(text){
  const out = [], seen = {};
  const hit = String(text||'').match(_MAIL_RE) || [];
  hit.forEach(h=>{ const e = normEmail(h); if(e && !seen[e]){ seen[e]=1; out.push(e); } });
  return out;
}
/* Firebase 키에는 . 을 쓸 수 없는데 이메일에는 반드시 있다.
   저장할 때 쉼표로 바꿔 넣고 읽을 때 되돌린다. 이메일 주소에는 쉼표가 올 수 없어 안전하다. */
function emailKey(e){ return String(e||'').replace(/\./g, ','); }
function keyEmail(k){ return String(k||'').replace(/,/g, '.'); }
/* 보낼 사람을 고르고, 왜 몇 명이 빠졌는지 함께 돌려준다.
   몇 명에게 나가는지 모르고 보내는 일이 없어야 한다. */
function mailTargets(items, block){
  const B = block || {};
  const ok = [], seen = {};
  let noEmail=0, blocked=0, dup=0;
  (items||[]).forEach(it=>{
    if(!it) return;
    if(inLockedGroup(it)) return;              /* 잠긴 폴더 명함은 단체 메일에 섞이면 안 된다 */
    const e = normEmail(it.email);
    if(!e){ noEmail++; return; }
    if(it.noMail || B[emailKey(e)]){ blocked++; return; }
    if(seen[e]){ dup++; return; }
    seen[e] = 1;
    ok.push({ name:it.name||'', email:e, company:it.company||'',
              title:it.title||'', mobile:it.mobile||it.companyTel||it.tel||'' });
  });
  return { ok, stat:{ ready:ok.length, noEmail, blocked, dup } };
}
/* CSV — 엑셀·주소록이 한글을 제대로 읽게 BOM 을 붙이고, 값에 든 쉼표·따옴표·줄바꿈을 감싼다. */
const CSV_COLS = [['name','이름'],['email','이메일'],['company','회사'],['title','직책'],['mobile','휴대폰']];
function _csvCell(v){
  const s = String(v==null?'':v);
  return /[",\n\r]/.test(s) ? '"' + s.replace(/"/g,'""') + '"' : s;
}
function toCsv(rows){
  const head = CSV_COLS.map(c=>c[1]).join(',');
  const body = (rows||[]).map(r=> CSV_COLS.map(c=>_csvCell(r[c[0]])).join(',')).join('\r\n');
  return '\uFEFF' + head + '\r\n' + body + '\r\n';
}

/* 선택한 명함만 대상으로 한다.
   ★ 엑셀 내보내기와 달리 '아무것도 안 고르면 전체'로 넘어가지 않는다 —
     실수로 누르면 6,275명 주소가 통째로 나간다. */
function selMailPick(){
  const ids = Object.keys(state.sel);
  if(!ids.length){ toast('명함을 먼저 선택하세요'); return null; }
  const items = ids.map(id=>state.items[id]).filter(Boolean);
  const r = mailTargets(items, state.mailBlock);
  if(!r.ok.length){
    toast(`보낼 수 있는 주소가 없습니다 — 이메일 없음 ${r.stat.noEmail} · 수신거부 ${r.stat.blocked}`, 6000);
    return null;
  }
  return r;
}
/* 결과를 숫자로 알린다. 몇 명에게 나가는지 모르고 보내면 안 된다. */
function _mailStatText(st){
  const bits = [];
  if(st.noEmail) bits.push(`이메일 없음 ${st.noEmail}`);
  if(st.blocked) bits.push(`수신거부 ${st.blocked}`);
  if(st.dup)     bits.push(`중복 ${st.dup}`);
  return `${st.ready.toLocaleString()}명 준비됨` + (bits.length ? ` · ${bits.join(' · ')} 제외` : '');
}
